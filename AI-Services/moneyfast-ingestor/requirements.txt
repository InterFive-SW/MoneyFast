# MoneyFast ingestor runtime deps.
#
# No database driver here on purpose: this service holds no DB credential and
# pushes everything over HTTP, which is what keeps base-db and vector-db
# isolated from the one component that reaches the internet.

httpx>=0.27,<0.29
apscheduler>=3.10,<4

# Market data. yfinance pulls pandas; that is the cost of getting OHLCV
# history without an API key, and history is what makes real risk metrics
# possible at all.
yfinance>=0.2.40,<0.3

# RSS. feedparser handles the four feed dialects Turkish outlets use between
# them; parsing them by hand is a bug farm.
feedparser>=6.0,<7

# KAP disclosure attachments. Financial statements are filed as PDFs, and the
# body of a quarterly report is entirely in the attachment. Same library the
# backend already uses for user uploads (`documents.py`), so this adds a
# consumer rather than a new dependency to the system.
pypdf>=5.0.0,<7
